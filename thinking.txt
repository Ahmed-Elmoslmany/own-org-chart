Convert ORM structure to Org chart structure
const buildOrgChartStructure = (nodes) => {
  const hashMap = {};
  const orgChart = [];
  
  for (const node of nodes) {
      hashMap[node.id] = { ...node, children: [] }; // Hashing nodes
  }
  
  for (const node of nodes) {
      if (node.parentId === null) {
        orgChart.push(hashMap[node.id]);
      } else {
        const parent = hashMap[node.parentId];
        parent.children.push(hashMap[node.id]);
      }
  }
  return orgChart;
}
